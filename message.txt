local sel = "Nerd"
local ishold = false
local tgt = 0
local isauto = false
local firing = false
local wasact = false

local function getcount()
    local f = workspace:FindFirstChild("Nextbots")
    return f and #f:GetChildren() or 0
end

local function gettally()
    local t = {}
    local f = workspace:FindFirstChild("Nextbots")
    if f then
        for _, m in ipairs(f:GetChildren()) do
            t[m.Name] = (t[m.Name] or 0) + 1
        end
    end
    return t
end

local function isact()
    local r = vals:FindFirstChild("roundActive")
    return r and r.Value == true
end

local function getstat()
    local s = vals:FindFirstChild("status")
    return s and s.Value or "?"
end

local function gettime()
    local t = vals:FindFirstChild("timer")
    return t and t.Value or "--"
end

local function getnum()
    local n = tonumber(amt.Text)
    if not n then return 0 end
    n = math.floor(n)
    if n < 1 then return 0 end
    if n > limit then n = limit end
    return n
end

local function updscl()
    for n, b in pairs(btns) do
        if n == sel then
            b.TextColor3 = Color3.fromRGB(255, 255, 255)
            b.UIStroke.Color = accent
        else
            b.TextColor3 = Color3.fromRGB(200, 200, 200)
            b.UIStroke.Color = Color3.fromRGB(60, 60, 65)
        end
    end
end

local function getpick(i, mode)
    if mode == "Mix" then
        return types[((i - 1) % #types) + 1]
    elseif mode == "Rand" then
        return types[math.random(1, #types)]
    end
    return mode
end

local function dosummon(n, mode)
    if firing then return false end
    if not isact() or getcount() == 0 then
        stat.Text = "blocked: round is dead"
        return false
    end
    
    firing = true
    task.spawn(function()
        local s = 0
        while s < n do
            local c = math.min(chunk, n - s)
            for i = 1, c do
                local p = getpick(s + i, mode)
                pcall(function() remote:FireServer(p) end)
            end
            s = s + c
            if s < n then runs.Heartbeat:Wait() end
        end
        firing = false
    end)
    return true
end

exe.MouseButton1Click:Connect(function()
    local n = getnum()
    if n > 0 then dosummon(n, sel) end
end)

for n, b in pairs(btns) do
    b.MouseButton1Click:Connect(function()
        sel = n
        updscl()
        if ishold then
            tgt = getnum()
            holdbtn.Text = "Hold: " .. tgt
        end
    end)
end

    holdbtn.Text = ishold and ("Hold: " .. tgt) or "Hold: Off"
    holdstroke.Color = ishold and accent or Color3.fromRGB(60, 60, 65)

uis.InputBegan:Connect(function(inp, typ)
    if typ then return end
    if inp.KeyCode == Enum.KeyCode.RightShift then
        gui.Enabled = not gui.Enabled
    end
end)

task.spawn(function()
    while task.wait(delay) do
        local act = isact()
        
        if isauto and act and not wasact then
            dosummon(getnum(), sel)
        end
        wasact = act
        
        if ishold and tgt > 0 and not firing and act then
            local req = math.min(chunk, tgt - getcount())
            if req > 0 then
                for i = 1, req do
                    local p = getpick(i, sel)
                    pcall(function() remote:FireServer(p) end)
                end
            end
        end
    end
end)

task.spawn(function()
    while task.wait(0.3) do
        local t = gettally()
        local c = getcount()
        for n, b in pairs(btns) do
            local num = t[n]
            b.Text = n .. (num and " (" .. num .. ")" or "")
        end
        
        local act = isact()
        if act then
            stat.Text = "Live | Time: " .. gettime() .. " | Count: " .. c .. " | Sel: " .. sel
        else
            stat.Text = "Idle | Status: " .. getstat() .. " | Count: " .. c .. " | Sel: " .. sel
        end
    end
end)

    if inp.UserInputType == Enum.UserInputType.MouseButton1 or inp.UserInputType == Enum.UserInputType.Touch then
        drag = true
        start = inp.Position
        pos = bg.Position
        inp.Changed:Connect(function()
            if inp.UserInputState == Enum.UserInputState.End then
                drag = false
            end
        end)
    end
end)

    if inp.UserInputType == Enum.UserInputType.MouseMovement or inp.UserInputType == Enum.UserInputType.Touch then
        dragin = inp
    end
end)

    if drag and inp == dragin then
        local del = inp.Position - start
        bg.Position = UDim2.new(pos.X.Scale, pos.X.Offset + del.X, pos.Y.Scale, pos.Y.Offset + del.Y)
    end
end)
