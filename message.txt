local GENV = (type(getgenv) == "function" and getgenv()) or _G
if GENV.YY_UnlockAll and GENV.YY_UnlockAll.Disable then
    pcall(GENV.YY_UnlockAll.Disable)
end

GENV.YY_UnlockAll = (function()
    local M = {}
    local Players = game:GetService("Players")
    local RS = game:GetService("ReplicatedStorage")
    local StarterGui = game:GetService("StarterGui")
    local LP = Players.LocalPlayer
    local PREFIX = "[Unlock All]"
    local LOG = {}

    local function notify(title, content, sub, dur)
        pcall(function()
            local text = tostring(content or "")
            if sub and sub ~= "" then
                text = text .. "\n" .. tostring(sub)
            end
            StarterGui:SetCore("SendNotification", {
                Title = tostring(title or "Unlock All"),
                Text = text,
                Duration = dur or 5,
            })
        end)
    end

    local function log(...)
        local parts = {}
        for i = 1, select("#", ...) do
            parts[i] = tostring((select(i, ...)))
        end
        local line = PREFIX .. " " .. table.concat(parts, " ")
        LOG[#LOG + 1] = line
        if #LOG > 200 then
            table.remove(LOG, 1)
        end
        print(line)
    end

    local function path(root, ...)
        local cur = root
        for _, name in ipairs({ ... }) do
            if not cur then return nil end
            cur = cur:FindFirstChild(name)
        end
        return cur
    end

    local function req(label, ...)
        local names = { ... }
        local ok, res = pcall(function()
            local inst = path(RS, table.unpack(names))
            if not inst then error("path not found") end
            return require(inst)
        end)
        if not ok then
            log("FAILED loading", label, "->", tostring(res))
            return nil
        end
        return res
    end

    local active = false
    local gen = 0
    local patches = {}
    local patchResults = {}
    local fakeE = {}
    local fakeC = {}
    local pending = nil
    local popupInst = nil
    local warnedOverlay = false
    local lastBuild = {}
    local emoteOn, cosOn = false, false

    local Registry = req("Items.Registry", "Items", "Registry")
    local OwnedBitset = req("OwnedBitset", "Shared", "UserData", "Codecs", "OwnedBitset")
    local EquippedCodec = req("EquippedCodec", "Shared", "UserData", "Codecs", "Equipped")
    local Store = req("Store", "Client", "Services", "UserData", "Inventory", "Store")

    local RigClass = req("Objects.Game.Character.Rig", "Objects", "Game", "Character", "Rig")
    local DataRegClass = req("Objects.Game.Character.DataRegistry", "Objects", "Game", "Character", "DataRegistry")
    local PopupService = req("PopupService", "Services", "UI", "PopupService")
    local EmotePopup = req("PopupService.Types.Emote", "Services", "UI", "PopupService", "Types", "Emote")
    local EmoteClass = req("Objects.Items.Emote", "Objects", "Items", "Emote")
    local ItemSvc = req("ClientItemService", "Services", "Items", "ClientItemService")

    local InvView
    pcall(function()
        InvView = require(RS.Objects.Menu.Menu.Views.Inventory)
    end)

    local TYPE_NAMES = { "Cosmetic", "Character", "Unusual", "Perk", "CarryAnimation", "Nametag" }
    local ENABLED, KNOWN = {}, {}
    for _, n in ipairs(TYPE_NAMES) do
        ENABLED[n] = true
        KNOWN[n] = true
    end
    local INGAME = { Perk = true, Nametag = true }

    local function emptyByType()
        local t = {}
        for _, n in ipairs(TYPE_NAMES) do
            t[n] = {}
        end
        return t
    end

    local BYTYPE = emptyByType()
    local TYPE_OF = {}
    local ALL, ALLSET = {}, {}

    M.Log = LOG
    M.FakeEmotes = fakeE
    M.FakeCosmetics = fakeC
    M.LastBuild = lastBuild

    local function getFieldStore()
        return Store and Store._fieldStore
    end

    local function getCache()
        local fs = getFieldStore()
        return fs and fs._cache
    end

    local function refreshUI()
        if InvView then
            if InvView.UpdateInventory then task.spawn(InvView.UpdateInventory, InvView) end
            if InvView.UpdateLoadout then task.spawn(InvView.UpdateLoadout, InvView) end
        end
    end

    local function patch(label, tbl, key, build)
        local orig = tbl[key]
        if type(orig) ~= "function" then
            log("PATCH NOT APPLIED (function does not exist):", label)
            patchResults[#patchResults + 1] = { label = label, ok = false }
            return false
        end
        patches[#patches + 1] = { tbl = tbl, key = key, orig = orig }
        tbl[key] = build(orig)
        patchResults[#patchResults + 1] = { label = label, ok = true }
        return true
    end

    local function realSlotId(idx)
        local c = getCache()
        if c and c.Equipped and EquippedCodec then
            local slotIdx = 12 + idx
            return EquippedCodec.Get(c.Equipped, slotIdx) or 0
        end
        return 0
    end

    local function getPopup()
        if popupInst and popupInst.Frame then
            return popupInst
        end
        if PopupService and type(PopupService.GetPopup) == "function" then
            local ok, p = pcall(PopupService.GetPopup, PopupService, "Emote")
            if ok and p then
                popupInst = p
                return p
            end
        end
        return nil
    end

    local function wheelSlotFrame(popup, idx)
        local wheel = popup.Frame:FindFirstChild(idx <= 6 and "Wheel" or "Wheel2")
        return wheel and wheel:FindFirstChild("Emote" .. ((idx - 1) % 6 + 1))
    end

    local function applyOverlay(popup)
        if not popup or not popup.Frame or type(popup.Emotes) ~= "table" or not ItemSvc then
            return
        end
        for idx, fid in pairs(fakeE) do
            if popup.Emotes[idx] then
                local ok, err = pcall(function()
                    local sf = wheelSlotFrame(popup, idx)
                    local item = ItemSvc:GetItemFromID(fid)
                    if sf and item then
                        ItemSvc:ClearViewport(sf.ViewportFrame)
                        ItemSvc:CreateViewport(item, sf.ViewportFrame)
                        sf.TextLabel.Text = ItemSvc:GetName(item)
                    end
                end)
                if not ok and not warnedOverlay then
                    warnedOverlay = true
                    log("Warning: could not draw visual in wheel ->", tostring(err))
                end
            end
        end
    end

    local function refreshWheel()
        local p = getPopup()
        if not p or type(p.Emotes) ~= "table" then
            return
        end
        applyOverlay(p)
    end

    local function isLocalEmote(self)
        local char = self.Character
        local lc = LP.Character
        if not char or not lc then
            return false
        end
        if char == lc then
            return true
        end
        local a, b = char:GetAttribute("Tag"), lc:GetAttribute("Tag")
        return a ~= nil and a == b
    end

    local function resolveFake(self)
        local mod = self.EmoteModule
        local modId = mod and mod:GetAttribute("ID")
        if not modId then
            return nil
        end
        if pending and tick() - pending.t < 5 then
            local fid = fakeE[pending.slot]
            if fid and realSlotId(pending.slot) == modId then
                pending = nil
                return fid
            end
        end
        for idx = 1, 12 do
            local fid = fakeE[idx]
            if fid and realSlotId(idx) == modId then
                return fid
            end
        end
        return nil
    end

    local function swapTo(fid)
        if type(firesignal) ~= "function" then
            log("Executor has no firesignal: local emote cannot be changed")
            return
        end
        local char = LP.Character
        local tag = char and char:GetAttribute("Tag")
        if not tag then
            log("Character tag not found; emote not swapped.")
            return
        end
        local ok, err = pcall(function()
            local tagBuf = buffer.create(2)
            buffer.writeu16(tagBuf, 0, tag)
            firesignal(RS.Events.UpdateCharacterDataRegistry.OnClientEvent, {
                buffer.fromstring("\24\1"),
                fid,
                tagBuf,
            })
        end)
        if not ok then
            log("Local emote swap failed ->", tostring(err))
        end
    end

    local function collectItems()
        if not Registry then return end
        local okAll, all = pcall(Registry.GetAll)
        if not okAll or type(all) ~= "table" then return end

        local rawEmotes = {}
        local bytype = emptyByType()
        local typeOf = {}
        local count = 0

        for id, item in pairs(all) do
            if type(id) == "number" then
                count = count + 1
                local cats = item.Categories or {}
                for _, cat in ipairs(cats) do
                    if cat == "Emotes" then
                        rawEmotes[#rawEmotes + 1] = id
                        break
                    end
                    if KNOWN[cat] then
                        local list = bytype[cat]
                        list[#list + 1] = id
                        typeOf[id] = cat
                    end
                end
            end
        end

        table.sort(rawEmotes)
        for _, list in pairs(bytype) do
            table.sort(list)
        end

        return rawEmotes, bytype, typeOf, count
    end

    local function substitutePerks(value)
        if type(value) ~= "table" or not (INGAME.Perk and ENABLED.Perk) then
            return value
        end
        local out = table.clone(value)
        for _, fid in pairs(fakeC) do
            if TYPE_OF[fid] == "Perk" then
                table.insert(out, fid)
            end
        end
        return out
    end

    local function substituteNametag(value)
        local fid = fakeC["NametagSlot"]
        if type(fid) == "number" and INGAME.Nametag and ENABLED.Nametag then
            return fid
        end
        return value
    end

    local function unlockAllInCache()
        local c = getCache()
        if not (c and c.OwnedBits and OwnedBitset) then return end
        local cur = c.OwnedBits
        for id in pairs(ALLSET) do
            cur = OwnedBitset.Add(cur, id)
        end
        c.OwnedBits = cur
        local fs = getFieldStore()
        if fs and fs._listeners and fs._listeners.OwnedBits then
            for _, l in ipairs(fs._listeners.OwnedBits) do
                pcall(l, c.OwnedBits)
            end
        end
        refreshUI()
        log("Injected items into live OwnedBits cache")
    end

    local function applyPatches()
        if emoteOn and EmotePopup then
            patch("EmotePopup.Show", EmotePopup, "Show", function(orig)
                return function(self, ...)
                    local r = orig(self, ...)
                    popupInst = self
                    applyOverlay(self)
                    return r
                end
            end)
            patch("EmotePopup.KeyActivated", EmotePopup, "KeyActivated", function(orig)
                return function(self, slot, ...)
                    if type(slot) == "number" then
                        pending = { slot = slot, t = tick() }
                    end
                    return orig(self, slot, ...)
                end
            end)
        end

        if emoteOn and EmoteClass and EmoteClass.Activate then
            patch("Emote.Activate", EmoteClass, "Activate", function(orig)
                return function(self, preview, p3, ...)
                    local fid = nil
                    if not preview then
                        pcall(function()
                            if isLocalEmote(self) then
                                fid = resolveFake(self)
                            end
                        end)
                    end
                    if fid then
                        task.defer(swapTo, fid)
                    end
                    return orig(self, preview, p3, ...)
                end
            end)
        end

        if cosOn and RigClass and RigClass.BuildRig then
            patch("Rig.BuildRig", RigClass, "BuildRig", function(orig)
                return function(self, ...)
                    if not self.Built and type(self.Cosmetics) == "table" and next(fakeC) ~= nil then
                        local isLocal = false
                        pcall(function()
                            isLocal = self.Parent ~= nil and self.Parent.Local == true
                        end)
                        if isLocal then
                            pcall(function()
                                local real = self.Cosmetics
                                local out, seen = {}, {}
                                for _, id in ipairs(real) do
                                    out[#out + 1] = id
                                    seen[id] = true
                                end
                                for _, fid in pairs(fakeC) do
                                    if not seen[fid] then
                                        out[#out + 1] = fid
                                        seen[fid] = true
                                    end
                                end
                                self.Cosmetics = out
                                log("Rig built with visual cosmetics")
                            end)
                        end
                    end
                    return orig(self, ...)
                end
            end)
        end

        if cosOn and DataRegClass and DataRegClass.SetTerm then
            patch("DataRegistry.SetTerm", DataRegClass, "SetTerm", function(orig)
                return function(self, term, value, ...)
                    if (term == "Perks" or term == "Nametag") and next(fakeC) ~= nil then
                        pcall(function()
                            if self.Parent and self.Parent.Local == true then
                                if term == "Perks" then
                                    value = substitutePerks(value)
                                elseif term == "Nametag" then
                                    value = substituteNametag(value)
                                end
                            end
                        end)
                    end
                    return orig(self, term, value, ...)
                end
            end)
        end
    end

    local TYPE_LABELS = { Cosmetic = "cosmetics", Character = "characters", Unusual = "unusuals", Perk = "perks", CarryAnimation = "carry animations", Nametag = "nametags" }

    function M.Disable()
        gen = gen + 1
        if not active then return end
        for i = #patches, 1, -1 do
            local p = patches[i]
            pcall(function()
                p.tbl[p.key] = p.orig
            end)
        end
        patches = {}
        table.clear(fakeE)
        table.clear(fakeC)
        table.clear(lastBuild)
        pending = nil
        emoteOn, cosOn = false, false
        active = false
        refreshWheel()
        log("Unloaded: restored normal")
        notify("Unlock All disabled", "Everything was restored to normal.", nil, 4)
    end

    local function enableImpl(myGen)
        if not (Registry and OwnedBitset and EquippedCodec and Store) then
            log("ABORTED: required inventory/codec modules missing.")
            notify("Unlock All failed", "Required game modules not found.", nil, 5)
            return
        end

        local emotes, bytype, typeOf, total = collectItems()
        if not total or total == 0 then
            log("ABORTED: Registry returned no items.")
            return
        end

        ALL = emotes or {}
        ALLSET = {}
        for _, id in ipairs(ALL) do ALLSET[id] = true end
        for _, list in pairs(bytype or {}) do
            for _, id in ipairs(list) do ALLSET[id] = true end
        end

        BYTYPE = bytype or emptyByType()
        TYPE_OF = typeOf or {}
        emoteOn = #ALL > 0
        cosOn = total > #ALL

        patches, patchResults = {}, {}
        table.clear(fakeE)
        table.clear(fakeC)
        pending = nil
        popupInst = nil
        warnedOverlay = false

        applyPatches()
        unlockAllInCache()

        local fs = getFieldStore()
        if fs and fs._listeners and fs._listeners.OwnedBits then
            table.insert(fs._listeners.OwnedBits, function(newBuf)
                local cur = newBuf
                for id in pairs(ALLSET) do
                    if not OwnedBitset.Has(cur, id) then
                        cur = OwnedBitset.Add(cur, id)
                    end
                end
                local c = getCache()
                if c then c.OwnedBits = cur end
            end)
        end

        active = true
        log("Ready: Unlock All enabled cleanly without forced slots")
        notify("Unlock All enabled", total .. " items unlocked in inventory!", nil, 5)
    end

    function M.Enable()
        if active then return end
        gen = gen + 1
        local myGen = gen
        task.spawn(function()
            local ok, err = pcall(enableImpl, myGen)
            if not ok then
                log("Enable failed ->", tostring(err))
            end
        end)
    end

    function M.IsEnabled()
        return active
    end

    function M.SetVisualEmote(slotIndex, emoteId)
        fakeE[slotIndex] = emoteId
        log("Slot", slotIndex, "-> visual emote", emoteId)
        refreshWheel()
    end

    function M.SetVisualCosmetic(slotName, cosmeticId)
        fakeC[slotName] = cosmeticId
        log("Slot", slotName, "-> visual", cosmeticId)
    end

    function M.Status()
        log("---- STATUS ---- enabled:", tostring(active))
        for _, r in ipairs(patchResults) do
            log(r.ok and "OK     " or "FAILED ", r.label)
        end
        for idx = 1, 12 do
            if fakeE[idx] then
                log("Emote slot", idx, "visual:", fakeE[idx])
            end
        end
        for slot, fid in pairs(fakeC) do
            log("Cosmetic slot", slot, "visual:", fid)
        end
    end

    return M
end)()

task.spawn(pcall, GENV.YY_UnlockAll.Enable)
